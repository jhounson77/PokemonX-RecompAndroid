cmake_minimum_required(VERSION 3.25)
project(PokemonXMainAOT LANGUAGES CXX)
if(NOT ANDROID OR NOT PX_MAIN_SOURCE_DIR OR NOT PX_HOST_LIBRARY)
 message(FATAL_ERROR "Pass Android toolchain, PX_MAIN_SOURCE_DIR and PX_HOST_LIBRARY")
endif()
get_filename_component(HEXLITH_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../hexlith" ABSOLUTE)
add_subdirectory("${PX_MAIN_SOURCE_DIR}" generated)
add_library(px_host SHARED IMPORTED)
set_target_properties(px_host PROPERTIES IMPORTED_LOCATION "${PX_HOST_LIBRARY}")
target_link_libraries(hexlith_lifted PRIVATE px_host)
target_link_options(hexlith_lifted PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384)
set_target_properties(hexlith_lifted PROPERTIES OUTPUT_NAME pxmain15)
