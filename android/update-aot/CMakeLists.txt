cmake_minimum_required(VERSION 3.25)
project(PokemonXUpdateAOT LANGUAGES CXX)
if(NOT ANDROID OR NOT PX_UPDATE_SOURCE_DIR OR NOT PX_HOST_LIBRARY)
 message(FATAL_ERROR "Pass Android toolchain, PX_UPDATE_SOURCE_DIR and PX_HOST_LIBRARY")
endif()
get_filename_component(HEXLITH_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../hexlith" ABSOLUTE)
add_subdirectory("${PX_UPDATE_SOURCE_DIR}/main" main)
add_subdirectory("${PX_UPDATE_SOURCE_DIR}/modules" modules)
add_library(px_host SHARED IMPORTED)
set_target_properties(px_host PROPERTIES IMPORTED_LOCATION "${PX_HOST_LIBRARY}")
foreach(t hexlith_lifted hexlith_modules)
 target_link_libraries(${t} PRIVATE px_host)
 target_link_options(${t} PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384)
endforeach()
set_target_properties(hexlith_lifted PROPERTIES OUTPUT_NAME pxmain15)
set_target_properties(hexlith_modules PROPERTIES OUTPUT_NAME pxmodules15)
