cmake_minimum_required(VERSION 3.25)
project(PokemonXSupplementalAOT LANGUAGES CXX)
if(NOT ANDROID OR NOT PX_SUPPLEMENT_SOURCE_DIR OR NOT PX_HOST_LIBRARY)
 message(FATAL_ERROR "Pass Android toolchain, PX_SUPPLEMENT_SOURCE_DIR and PX_HOST_LIBRARY")
endif()
get_filename_component(HEXLITH_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../hexlith" ABSOLUTE)
add_subdirectory("${PX_SUPPLEMENT_SOURCE_DIR}" generated)
add_library(px_host SHARED IMPORTED)
set_target_properties(px_host PROPERTIES IMPORTED_LOCATION "${PX_HOST_LIBRARY}")
target_link_libraries(hexlith_modules PRIVATE px_host)
target_link_options(hexlith_modules PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384)
set_target_properties(hexlith_modules PROPERTIES OUTPUT_NAME pxintro15)
